rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function logado() { return request.auth != null; }
    function email() { return request.auth.token.email; }
    function ceo() { return logado() && email() == 'ceo@${SEG_DOM}'; }
    function adm() { return ceo() || (logado() && email() == 'admin@${SEG_DOM}'); }
    function chave() { return email().split('@')[0]; }
    function equipe() {
      return logado() && email().matches('^c[0-9]+@sistema-geceen[.]app$')
        && get(/databases/$(database)/documents/geceen/contas).data.get(chave(), false) == true;
    }
    function membro() { return adm() || equipe(); }
    function meuColab() { return int(chave().replace('^c', '')); }

    match /geceen/publico   { allow read: if true;     allow write: if adm(); }
    match /geceen/contas    { allow read: if logado(); allow write: if ceo(); }
    match /geceen/segredos  { allow read, write: if ceo(); }
    match /geceen/financeiro{ allow read, write: if adm(); }
    match /geceen/dados     { allow read: if membro(); allow write: if adm(); }
    match /geceen/cadastros { allow read, write: if membro(); }

    match /ordens/{id}  { allow read, create, update: if membro(); allow delete: if adm(); }
    match /osFotos/{id} { allow read, create, update: if membro(); allow delete: if adm(); }
    match /tarefas/{id} { allow read, create, update: if membro(); allow delete: if adm(); }
    match /frota/{id}   { allow read, create, update: if membro(); allow delete: if adm(); }
    match /estoque/{id} { allow read, create, update: if membro(); allow delete: if adm(); }
    match /cartaoDesp/{id} { allow read, create, update: if membro(); allow delete: if adm(); }
    match /etiquetas/{t} { allow get: if true; allow list: if false; allow write: if membro(); }
    match /portal/{t}    { allow get: if true; allow list: if false; allow write: if membro(); }
    match /site/{d}      { allow get: if true; allow list: if false; allow write: if adm(); }
    match /tv/{t}        { allow get: if true; allow list: if false; allow write: if membro(); }
    match /pesquisa/{t}  { allow get: if true; allow list: if false; allow write: if membro(); }
    match /respostas/{t} {
      allow create: if exists(/databases/$(database)/documents/pesquisa/$(t))
        && request.resource.data.nota is int && request.resource.data.nota >= 0 && request.resource.data.nota <= 10;
      allow get: if true; allow list: if membro(); allow update, delete: if adm();
    }
    match /orcamento/{t} { allow get: if true; allow list: if false; allow write: if membro(); }
    match /aprovacoes/{t} {
      allow create: if exists(/databases/$(database)/documents/orcamento/$(t))
        && request.resource.data.tok == t
        && request.resource.data.os == get(/databases/$(database)/documents/orcamento/$(t)).data.os
        && request.resource.data.decisao in ['aprovado','recusado']
        && request.resource.data.nome is string && request.resource.data.nome.size() < 120
        && (!('assin' in request.resource.data) || request.resource.data.assin.size() < 300000);
      allow get: if true; allow list: if membro(); allow update, delete: if adm();
    }
    match /chamados/{id} {
      allow create: if request.resource.data.status == 'novo' && request.resource.data.tok is string
        && request.resource.data.desc.size() < 2000
        && (exists(/databases/$(database)/documents/portal/$(request.resource.data.tok))
            || (request.resource.data.origem == 'site' && request.resource.data.tok == 'site' && !('cli' in request.resource.data)));
      allow read, update: if membro();
      allow delete: if adm();
    }

    match /batidas/{id} {
      allow read: if adm() || (equipe() && resource.data.colabId == meuColab());
      allow create: if adm() || (equipe() && request.resource.data.colabId == meuColab());
      allow update, delete: if adm();
    }
    match /batidaFotos/{id} {
      allow read: if adm() || (equipe() && resource.data.colabId == meuColab());
      allow create: if adm() || (equipe() && request.resource.data.colabId == meuColab());
      allow update, delete: if adm();
    }
    match /{tudo=**} { allow read, write: if false; }
  }
}
